    ```go
    package handlers

    // get_run is the GET /v1/runs/{id} handler. It projects a
    // runs.Meta + a paginated ReadOutput page into contract.RunDetail
    // (RUN-04 / OBS-03). The handler is read-only and does not write
    // to disk; redaction is already applied at read time by
    // runs.ReadOutput (17-04) — this handler only counts what came
    // back and emits the SEC-03 redaction.audit slog record per D-08.

    import (
        "encoding/json"
        "errors"
        "log/slog"
        "net/http"
        "strconv"

        "github.com/go-chi/chi/v5"

        "iac-runner/internal/contract"
        "iac-runner/internal/runs"
    )

    // GetRun returns the handler mounted at GET /v1/runs/{id} by
    // router.go. store is the runs.Store constructed in main.go at
    // startup (17-07).
    func GetRun(store *runs.Store) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            id := chi.URLParam(r, "id")

            // IsValidRunID guard runs BEFORE any filesystem access.
            // A hostile value like "../etc" must never reach
            // filepath.Join inside store.Load.
            if !runs.IsValidRunID(id) {
                writeError(w, http.StatusNotFound, contract.ErrCodeRunUnknownID,
                    "unknown run id \""+id+"\"",
                    "run ids are 16 base32 characters")
                return
            }

            meta, err := store.Load(id)
            if err != nil {
                if errors.Is(err, runs.ErrRunNotFound) {
                    writeError(w, http.StatusNotFound, contract.ErrCodeRunNotFound,
                        "no run with id \""+id+"\"", "")
                    return
                }
                // A non-ErrRunNotFound failure is a real filesystem
                // problem; surface as a generic 500 without leaking
                // the underlying error string.
                slog.Error("run_load_failed", "run_id", id, "err", err.Error())
                writeError(w, http.StatusInternalServerError, "internal",
                    "failed to load run; see iac-runner logs", "")
                return
            }

            // ?page= (default 1) and ?page_size= (default
            // contract.DefaultOutputPageSize, cap MaxOutputPageSize).
            q := r.URL.Query()
            page := parsePositiveInt(q.Get("page"), 1)
            pageSize := parsePositiveInt(q.Get("page_size"), contract.DefaultOutputPageSize)
            if pageSize > contract.MaxOutputPageSize {
                pageSize = contract.MaxOutputPageSize
            }

            pageResult, err := store.ReadOutput(id, page, pageSize)
            if err != nil {
                slog.Error("run_read_output_failed", "run_id", id, "err", err.Error())
                writeError(w, http.StatusInternalServerError, "internal",
                    "failed to read run output; see iac-runner logs", "")
                return
            }

            // D-08: emit redaction.audit only when redactions actually
            // happened. Operators watch this record for accidental
            // secret leakage (the count should track the number of
            // credential-shaped strings in tofu output).
            if pageResult.Redactions > 0 {
                slog.Info("iac_runner.redaction.audit",
                    "run_id", id,
                    "repo", meta.Repo,
                    "redactions", pageResult.Redactions,
                    "page", pageResult.Page,
                    "page_size", pageResult.PageSize,
                )
            }

            // Meta → RunDetail projection. StartedAt and FinishedAt
            // are time.Time on Meta but RFC3339 strings on RunDetail
            // (the wire shape — see contract.RunDetail's json tags),
            // so convert explicitly. ExitCode is *int on both sides;
            // a nil on Meta marshals as null on the wire (D-02).
            var startedAt, finishedAt string
            startedAt = meta.StartedAt.UTC().Format(time.RFC3339)
            if meta.FinishedAt != nil {
                finishedAt = meta.FinishedAt.UTC().Format(time.RFC3339)
            }

            detail := contract.RunDetail{
                RunID:       meta.RunID,
                Repo:        meta.Repo,
                Dir:         meta.Dir,
                Kind:        meta.Kind,
                Status:      meta.Status,
                ExitCode:    meta.ExitCode,
                StartedAt:   startedAt,
                FinishedAt:  finishedAt,
                ErrorCode:   meta.ErrorCode,
                OutputLines: pageResult.Lines,
                Page:        pageResult.Page,
                PageSize:    pageResult.PageSize,
                TotalLines:  pageResult.TotalLines,
            }

            w.Header().Set("Content-Type", "application/json")
            w.WriteHeader(http.StatusOK)
            _ = json.NewEncoder(w).Encode(detail)
        }
    }

    // parsePositiveInt returns the int parsed from s, or fallback if
    // s is empty / not a positive integer. The contract says page
    // and page_size are positive; a non-positive value is silently
    // replaced rather than rejected so a typo never blocks a
    // pagination read.
    func parsePositiveInt(s string, fallback int) int {
        if s == "" {
            return fallback
        }
        n, err := strconv.Atoi(s)
        if err != nil || n <= 0 {
            return fallback
        }
        return n
    }
    ```

    ```go
    package handlers

    // list_runs is the GET /v1/runs handler. RUN-05: returns the most
    // recent N runs newest-first (default 20, max 100), filtered by
    // ?repo= and ?status=. No output_lines — listing 20 runs must not
    // read 20 log files (the store.List contract reads meta.json
    // only; output is on-demand via GET /v1/runs/{id}).
    //
    // The handler is read-only; no slog record is emitted per call
    // (a polling client would otherwise flood the log stream).

    import (
        "encoding/json"
        "net/http"
        "strconv"

        "iac-runner/internal/contract"
        "iac-runner/internal/runs"
    )

    // ListRuns returns the handler mounted at GET /v1/runs by
    // router.go.
    func ListRuns(store *runs.Store) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            q := r.URL.Query()

            // ?repo= is a free-form string — the store's List filter
            // rejects unknown repos by returning no rows, which is the
            // right behavior (an empty list is not a 404 — the repo
            // may exist with zero runs yet).
            repo := q.Get("repo")

            // ?status= must match a RunStatus enum value. An unknown
            // value returns 400 + a clear message; the store would
            // otherwise never match.
            statusStr := q.Get("status")
            if statusStr != "" && !isKnownRunStatus(statusStr) {
                writeError(w, http.StatusBadRequest, "internal",
                    "invalid ?status= value \""+statusStr+"\"",
                    "use one of: queued, running, succeeded, failed, interrupted")
                return
            }

            // ?limit= is pre-clamped here so the response's Limit
            // field reflects what was actually applied. The store's
            // List also clamps, but the handler's clamp wins for
            // visibility.
            limit := contract.DefaultRunListLimit
            if v := q.Get("limit"); v != "" {
                if n, err := strconv.Atoi(v); err == nil && n > 0 {
                    limit = n
                }
            }
            if limit > contract.MaxRunListLimit {
                limit = contract.MaxRunListLimit
            }

            filter := runs.ListFilter{Repo: repo, Limit: limit}
            if statusStr != "" {
                filter.Status = contract.RunStatus(statusStr)
            }

            metas, err := store.List(filter)
            if err != nil {
                // The store only returns an error for filesystem
                // problems; surface as 500 without leaking detail.
                writeError(w, http.StatusInternalServerError, "internal",
                    "failed to list runs; see iac-runner logs", "")
                return
            }

            summaries := make([]contract.RunSummary, 0, len(metas))
            for _, m := range metas {
                var finishedAt string
                if m.FinishedAt != nil {
                    finishedAt = m.FinishedAt.UTC().Format(time.RFC3339)
                }
                summaries = append(summaries, contract.RunSummary{
                    RunID:      m.RunID,
                    Repo:       m.Repo,
                    Dir:        m.Dir,
                    Kind:       m.Kind,
                    Status:     m.Status,
                    ExitCode:   m.ExitCode,
                    StartedAt:  m.StartedAt.UTC().Format(time.RFC3339),
                    FinishedAt: finishedAt,
                    ErrorCode:  m.ErrorCode,
                })
            }

            w.Header().Set("Content-Type", "application/json")
            w.WriteHeader(http.StatusOK)
            _ = json.NewEncoder(w).Encode(contract.RunListResponse{
                Runs: summaries, Count: len(summaries), Limit: limit,
            })
        }
    }

    // isKnownRunStatus reports whether s matches one of the five
    // RunStatus enum values declared in contract/types.go (17-02).
    // The handler validates at the HTTP boundary so the store is
    // never asked for an unknown status string.
    func isKnownRunStatus(s string) bool {
        switch contract.RunStatus(s) {
        case contract.RunStatusQueued, contract.RunStatusRunning,
            contract.RunStatusSucceeded, contract.RunStatusFailed,
            contract.RunStatusInterrupted:
            return true
        }
        return false
    }
    ```

    ```go
    //
    // Phase 16 signature: (runnerVersion, store, validator). Phase 17
    // adds the queue + run store + git manager dependencies for the
    // five new endpoints (GIT-03, RUN-02/03/04/05). RUN-01 (the

    func NewRouter(
        runnerVersion string,
        store *auth.TokenStore,
        validator *keys.Validator,
        gitMgr *git.Manager,
        runStore *runs.Store,
        q *jobq.Queue,
    ) http.Handler {
        r := chi.NewRouter()

        r.Use(chimiddleware.RequestID)
        r.Use(chimiddleware.Recoverer)
        r.Use(reqlog.RequestLogger())

        // Public, unauthenticated.

        r.Get("/healthz", handlers.Healthz(runnerVersion, validator))

        // Auth-protected /v1/*.
        r.Route("/v1", func(r chi.Router) {

            r.Post("/auth/rotate", handlers.AuthRotate(store))
            r.Post("/repos/{name}/pull", handlers.ReposPull(gitMgr))
            r.Post("/plan", handlers.Plan(q))
            r.Post("/apply", handlers.Apply(q))
            r.Get("/runs/{id}", handlers.GetRun(runStore))
            r.Get("/runs", handlers.ListRuns(runStore))
        })

        return r
    }
    ```
