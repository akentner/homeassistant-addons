    ```go
    package handlers

    // repos_pull is the POST /v1/repos/{name}/pull handler. It is the
    // HTTP surface of git.Manager.Pull (17-03) — 17-03 owns the
    // GIT_SSH_COMMAND construction, the backoff curve, the stderr
    // classifier; this file owns the request parse, the typed-error
    // → HTTP mapping (D-19), and the response shape. RequireBearer
    // runs first (router.go), so a missing or wrong token never
    // reaches this code.
    //
    // The handler does NOT mutate any persistent state itself — Pull
    // does the work and returns; the handler is a translator.

    import (
        "encoding/json"
        "errors"
        "log/slog"
        "net/http"

        "github.com/go-chi/chi/v5"

        "iac-runner/internal/contract"
        "iac-runner/internal/git"
    )

    // reposPullResponse is the 200 body of POST /v1/repos/{name}/pull.
    // Mode is "ff-only" for a fast-forward pull on the configured
    // branch, "ref" for an explicit ref checkout (D-10/D-11). Head is
    // the post-pull commit SHA so the operator can pin the next plan
    // to the same state.
    type reposPullResponse struct {
        Name string `json:"name"`
        Mode string `json:"mode"`
        Head string `json:"head"`
    }

    // ReposPull returns the handler mounted at POST /v1/repos/{name}/pull
    // by router.go.
    func ReposPull(gitMgr *git.Manager) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            name := chi.URLParam(r, "name")

            cfg, ok := gitMgr.Repo(name)
            if !ok {
                writeError(w, http.StatusNotFound, contract.ErrCodeRunUnknownRepo,
                    "unknown repo \""+name+"\"",
                    "add it to the `repos` Options list")
                return
            }

            // ffOnly is the spec-default. A repo with a configured
            // `ref` overrides the git-side behavior (D-10); Manager.Pull
            // already inspects cfg.Ref and selects fetch+checkout vs
            // pull --ff-only, so the handler does NOT branch here.
            ffOnly := true

            outcome, err := gitMgr.Pull(r.Context(), name, ffOnly)
            if err != nil {
                writeGitError(w, err)
                return
            }

            slog.Info("iac_runner.repo.pulled",
                "repo", name, "url", cfg.URL, "mode", outcome.Mode, "head", outcome.Head,
            )

            w.Header().Set("Content-Type", "application/json")
            w.WriteHeader(http.StatusOK)
            _ = json.NewEncoder(w).Encode(reposPullResponse{
                Name: outcome.Name, Mode: outcome.Mode, Head: outcome.Head,
            })
        }
    }

    // writeGitError maps a *git.Error (D-19) to the matching HTTP
    // response. Codes are stable: ref_pull_incompatible is a
    // request-shape problem → 400; the rest are infrastructure
    // problems → 502. Any non-*git.Error is treated as a generic
    // git_clone_failed so the body never carries an opaque error
    // string or a stack trace.
    func writeGitError(w http.ResponseWriter, err error) {
        var ge *git.Error
        if errors.As(err, &ge) {
            status := http.StatusBadGateway
            if ge.Code == contract.ErrCodeGitRefPullIncompatible {
                status = http.StatusBadRequest
            }
            writeError(w, status, ge.Code, ge.Message, ge.Hint)
            return
        }
        writeError(w, http.StatusBadGateway, contract.ErrCodeGitCloneFailed,
            "git pull failed; see iac-runner logs", "")
    }
    ```

    B. Create `iac-runner/internal/httpapi/handlers/write_error.go` (shared with Task 2 here and with 17-08's GetRun / ListRuns handlers).

    ```go
    package handlers

    // write_error is the single helper every handler in this package
    // uses to emit a contract.ErrorResponse. Centralizing it guarantees
    // a uniform 4xx/5xx body shape across every Phase 17 endpoint and
    // keeps the per-handler code focused on the success path.

    import (
        "encoding/json"
        "net/http"

        "iac-runner/internal/contract"
    )

    // writeError writes the JSON error body with the supplied status
    // and the contract error_code. The hint is the operator-facing
    // Options-field fix suggested by the D-19 taxonomy; empty string
    // means "no actionable hint beyond the message".
    func writeError(w http.ResponseWriter, status int, code, message, hint string) {
        if hint != "" {
            // The hint is concatenated into message rather than a
            // separate field because contract.ErrorResponse carries
            // only error_code / message / request_id. Operators read
            // the JSON, see the hint as the last sentence of message,
            // and grep it.
            message = message + " — " + hint
        }
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(contract.ErrorResponse{
            ErrorCode: code, Message: message,
        })
    }
    ```

    ```go
    // /v1/apply (RUN-02 / RUN-03). The runner never reads anything
    // else from the body — the user-controlled fields are repo and
    // dir, validated by jobq.Submit.
    type planApplyRequest struct {
        Repo string `json:"repo"`
        Dir  string `json:"dir"`
    }

    // submitRequest translates the parsed HTTP body into the typed
    // jobq.Request. Kind is fixed by which handler is invoked.
    func submitRequest(body planApplyRequest, kind contract.RunKind) jobq.Request {
        return jobq.Request{Repo: body.Repo, Dir: body.Dir, Kind: kind}
    }

    // Plan returns the handler mounted at POST /v1/plan by router.go.
    func Plan(q *jobq.Queue) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            handleSubmit(w, r, q, contract.RunKindPlan)
        }
    }

    // Apply returns the handler mounted at POST /v1/apply by router.go.
    func Apply(q *jobq.Queue) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            handleSubmit(w, r, q, contract.RunKindApply)
        }
    }

    // handleSubmit is the shared body-parse + Submit + 202/503/typed-error
    // path. Two reasons not to inline: (1) Plan and Apply are
    // character-for-character identical apart from Kind; (2) keeping
    // the body parser in one place guarantees a malformed-body 400 has
    // the exact same shape from either endpoint.
    func handleSubmit(w http.ResponseWriter, r *http.Request, q *jobq.Queue, kind contract.RunKind) {
        var body planApplyRequest
        dec := json.NewDecoder(r.Body)
        dec.DisallowUnknownFields()
        if err := dec.Decode(&body); err != nil {
            writeError(w, http.StatusBadRequest, contract.ErrCodeRunInvalidDir,
                "request body must be JSON {\"repo\":\"<name>\",\"dir\":\"<subpath>\"}", "")
            return
        }

        runID, err := q.Submit(submitRequest(body, kind))
        if err != nil {
            // Capacity exhaustion is its own well-defined case —
            // it returns the bare sentinel jobq.ErrCapacityExhausted
            // (not a *git.Error). Every other Submit failure is a
            // typed *git.Error{Code: ErrCodeRunUnknownRepo |
            // ErrCodeGitCloneMissing | ErrCodeRunInvalidDir |
            // ErrCodeRunTofuNotFound}.
            if errors.Is(err, jobq.ErrCapacityExhausted) {
                w.Header().Set("Retry-After", "30")
                writeError(w, http.StatusServiceUnavailable,
                    contract.ErrCodeApplyCapacityExhausted,
                    "apply capacity exhausted; retry after 30s",
                    "raise `max_parallel_jobs` in Options or wait for in-flight jobs to finish")
                return
            }
            writeGitError(w, err)
            return
        }

        slog.Info("iac_runner.run.queued",
            "run_id", runID, "repo", body.Repo, "dir", body.Dir, "kind", string(kind),
        )

        w.Header().Set("Location", fmt.Sprintf("/v1/runs/%s", runID))
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(http.StatusAccepted)
        _ = json.NewEncoder(w).Encode(contract.RunAccepted{
            RunID: runID, Status: contract.RunStatusQueued,
        })
    }
    ```
