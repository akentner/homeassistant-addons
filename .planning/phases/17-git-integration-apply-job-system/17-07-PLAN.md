    // ─── Phase 17: Git manager + run store + retention + job queue ───

    // Build the git manager. NewManager takes the reposDir
    // (/data/repos — created lazily by CloneAll), the keysDir
    // (/data/keys, already chmod-600-validated by the keys
    // validator above), the parsed repo configs, and the default
    // CommandRunner + sleep injected by 17-03. A construction error
    // is fatal — without a manager, /v1/repos/{name}/pull would
    // never know which repos exist.
    gitMgr, err := git.NewManager(
        defaultDataDir+"/repos", // /data/repos
        defaultKeysDir,          // /data/keys (already 0600)
        opts.Repos,
        nil, // 17-03 default exec.Command runner
        nil, // 17-03 default time.Sleep
    )
    if err != nil {
        slog.Error("git_manager_init_failed", "err", err.Error())
        os.Exit(1)
    }
    slog.Info("repos_loaded",
        "count", len(opts.Repos),
        "repos", repoNames(opts.Repos),
    )

    // Best-effort startup clone. CloneAll never returns an error;
    // per-repo outcomes are logged here, NOT in NewManager, so a
    // failed clone is operator-visible (D-14 — non-fatal at boot).
    for _, outcome := range gitMgr.CloneAll(context.Background()) {
        if outcome.Err != nil {
            slog.Warn("git_clone_failed",
                "repo", outcome.Name,
                "attempts", outcome.Attempts,
                "err", outcome.Err.Error(),
            )
        } else if !outcome.Skipped {
            slog.Info("git_clone_succeeded",
                "repo", outcome.Name,
                "attempts", outcome.Attempts,
            )
        }
    }

    // Build the run store. NewStore does os.MkdirAll(runsDir, 0o700)
    // so /data/runs exists by the time SweepInterrupted runs.
    runStore, err := runs.NewStore(defaultDataDir+"/runs", nil) // nil now → time.Now
    if err != nil {
        slog.Error("runs_store_init_failed", "err", err.Error())
        os.Exit(1)
    }

    // SweepInterrupted (D-02): every run still marked `running`
    // from a previous container is rewritten to `interrupted` so
    // an operator's GET /v1/runs shows a stable post-mortem rather
    // than a perpetual running row.
    if n, err := runStore.SweepInterrupted(); err != nil {
        slog.Warn("runs_sweep_interrupted_failed", "err", err.Error())
    } else if n > 0 {
        slog.Info("runs_swept_interrupted", "count", n)
    }

    // Retention ticker (D-25): every runs_retention_hours / 4 the
    // ticker deletes run directories older than runs_retention_hours.
    // stopRetention is called from signals.go inside the SIGTERM
    // drain so the ticker doesn't outlive the process.
    stopRetention := runStore.StartRetentionTicker(
        context.Background(),
        time.Duration(opts.RunsRetentionHours)*time.Hour,
    )
    defer stopRetention() // belt-and-braces; signals.go calls it too

    // Build the job queue. jobq.New's Deps contract: nil Exec →
    // DefaultExec (os/exec with SIGTERM→SIGKILL), nil Now →
    // time.Now, TofuPath "" → resolved via exec.LookPath at
    // construction (a lookup failure is non-fatal — Submit returns
    // run_tofu_not_found and the runner still starts).
    q, err := jobq.New(jobq.Deps{
        Store:        runStore,
        Git:          gitMgr,
        Backend:      backend,
        MaxParallel:  opts.MaxParallelJobs,
        ApplyTimeout: time.Duration(opts.ApplyTimeoutMinutes) * time.Minute,
    })
    if err != nil {
        slog.Error("jobq_init_failed", "err", err.Error())
        os.Exit(1)
    }
    slog.Info("jobq_ready",
        "max_parallel_jobs", opts.MaxParallelJobs,
        "apply_timeout_minutes", opts.ApplyTimeoutMinutes,
    )
