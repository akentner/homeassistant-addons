class OriginStore:
    def __init__(self, path: Path) -> None
    def get_user(self, name: str) -> str                      # "gui" when unknown
    def get_acl(self, user: str, topic: str) -> str
    def get_token(self, user: str, token_id_: str) -> str
    def set_user(self, name: str, origin: str) -> None
    def set_acl(self, user: str, topic: str, origin: str) -> None
    def set_token(self, user: str, token_id_: str, origin: str) -> None
    def drop_user(self, name: str) -> None                    # cascades acls, tokens, stash
    def drop_acl(self, user: str, topic: str) -> None
    def drop_token(self, user: str, token_id_: str) -> None
    def clear_acls(self) -> None
    def clear_all(self) -> None
    def stash_acl(self, user: str, rules: list[dict]) -> None   # rules: {"topic","permission","origin"}
    def pop_acl_stash(self, user: str) -> list[dict]
    def prune(self, live_users: set[str], live_acls: set[tuple[str, str]], live_tokens: set[tuple[str, str]]) -> int
    def snapshot(self) -> dict                                # deep copy of the loaded document
class BackupStore:
    def __init__(self, backup_dir: Path, auth_db: Path, keep: int, clock: Callable[[], float] = time.time) -> None
    def create(self, reason: str = "manual") -> Backup
    def list(self) -> list[Backup]
    def path_for(self, name: str) -> Path
    def rotate(self) -> int
    def create_daily(self) -> Backup | None
def build_export(db: AuthReader, origins: OriginStore, *, include_hashes: bool, auth_default_access: str,
                 now: float) -> dict
def to_json(doc: dict) -> str
def to_yaml(doc: dict) -> str
