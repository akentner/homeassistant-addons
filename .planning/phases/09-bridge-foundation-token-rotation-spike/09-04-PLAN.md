    #!/usr/bin/env bash
    # spike-h1-token-rotation.sh — Phase 9 H-1 (PITFALLS): does SUPERVISOR_TOKEN
    # rotate across a Supervisor restart? Empirical evidence per CONTEXT.md D-15.
    #
    # PRE-APPROVED in /gsd-discuss-phase. AGENTS.md Live Systems rule: each
    # live-system action is announced in the executor's output BEFORE the
    # command runs.
    #
    # Usage: ./internal/spike-h1-token-rotation.sh [haos-host]
    # Default host: haos-op3050-1

    set -euo pipefail

    HOST="${1:-haos-op3050-1}"
    RUN_ID="$(date -u +%Y%m%dT%H%M%SZ)-$$"
    TRANSCRIPT="/tmp/spike-h1-${RUN_ID}.log"
    STUB_SLUG="terraform-bridge-spike"

    exec > >(tee -a "${TRANSCRIPT}") 2>&1

    echo "H-1 SPIKE — Phase 9 SUPERVISOR_TOKEN rotation across Supervisor restart"
    echo "Host: ${HOST}    Run ID: ${RUN_ID}    Transcript: ${TRANSCRIPT}"
    echo

    # ── Step 1: build the stub image locally, push to host, install via Supervisor API ─
    echo "── Step 1: build & ship stub terraform-bridge image ──"
    docker build -t "terraform-bridge-spike:latest" \
                 --build-arg "BRIDGE_VERSION=0.1.0" \
                 terraform-bridge/
    docker save "terraform-bridge-spike:latest" | ssh "${HOST}" docker load
    echo

    # Build the add-on's config locally and install via Supervisor API (not the
    # HA store — we're testing the Supervisor → add-on token injection path,
    # not the store install path).
    cat > /tmp/${STUB_SLUG}.yaml <<EOF
    name: "Spike Stub"
    description: "H-1 spike stub for SUPERVISOR_TOKEN rotation test"
    version: "0.0.1-0"
    slug: "${STUB_SLUG}"
    arch: [amd64]
    url: "https://github.com/akentner/homeassistant-addons"
    startup: "application"
    boot: "manual"
    host_network: false
    hassio_api: true
    hassio_role: manager
    ports: {}
    options: {}
    schema: {}
    EOF

    echo "── Step 2: install stub via Supervisor API (writes config to /tmp/${STUB_SLUG}.yaml and POSTs) ──"
    # The Supervisor API expects the addon's slug + an image reference; this is
    # the registry-less install-on-supervisor path. The exact API call is
    # `POST /api/store/repositories` + `POST /api/addons/${slug}/install`; the
    # spike uses `ha addons install` over SSH for simplicity.
    scp /tmp/${STUB_SLUG}.yaml "${HOST}:/tmp/"
    ssh "${HOST}" sudo ha addons install "local/spike:latest" || true
    sleep 3
    ssh "${HOST}" sudo ha addons start "${STUB_SLUG}"
    sleep 5
    echo

    # ── Step 3: capture SUPERVISOR_TOKEN from inside the stub container ──
    echo "── Step 3: capture SUPERVISOR_TOKEN BEFORE Supervisor restart ──"
    BEFORE_TOKEN=$(ssh "${HOST}" sudo ha addons stdin "${STUB_SLUG}" sh -c 'tr "\\0" "\\n" < /proc/1/environ | grep "^SUPERVISOR_TOKEN="' | sed 's/^SUPERVISOR_TOKEN=//')
    BEFORE_FP=$(printf '%s' "${BEFORE_TOKEN}" | sha256sum | cut -d' ' -f1)
    BEFORE_TS=$(date -u +%FT%TZ)
    echo "Captured before token (length=$(printf '%s' "${BEFORE_TOKEN}" | wc -c), sha256=${BEFORE_FP}) at ${BEFORE_TS}"

    # ── Step 4: Supervisor restart ──
    echo
    echo "── Step 4: ha supervisor restart (service-disrupting — announced above) ──"
    ssh "${HOST}" sudo ha supervisor restart
    echo "Supervisor restart issued at $(date -u +%FT%TZ)"
    echo "Waiting 60s for Supervisor to come back + add-on respawn + token re-injection"
    sleep 60
    echo

    # ── Step 5: capture SUPERVISOR_TOKEN again ──
    echo "── Step 5: capture SUPERVISOR_TOKEN AFTER Supervisor restart ──"
    ssh "${HOST}" sudo ha addons start "${STUB_SLUG}" 2>&1 || true
    sleep 5
    AFTER_TOKEN=$(ssh "${HOST}" sudo ha addons stdin "${STUB_SLUG}" sh -c 'tr "\\0" "\\n" < /proc/1/environ | grep "^SUPERVISOR_TOKEN="' | sed 's/^SUPERVISOR_TOKEN=//')
    AFTER_FP=$(printf '%s' "${AFTER_TOKEN}" | sha256sum | cut -d' ' -f1)
    AFTER_TS=$(date -u +%FT%TZ)
    echo "Captured after  token (length=$(printf '%s' "${AFTER_TOKEN}" | wc -c), sha256=${AFTER_FP}) at ${AFTER_TS}"

    # ── Step 6: compare ──
    echo
    echo "── Step 6: compare fingerprints ──"
    if [[ "${BEFORE_FP}" == "${AFTER_FP}" ]]; then
        RESULT="token_unchanged"
    else
        RESULT="token_rotated"
    fi
    echo "  before sha256: ${BEFORE_FP}  at ${BEFORE_TS}"
    echo "  after  sha256: ${AFTER_FP}   at ${AFTER_TS}"
    echo "RESULT: ${RESULT}"
    echo
    echo "Time delta before/after capture: $(($(date -d "${AFTER_TS}" +%s) - $(date -d "${BEFORE_TS}" +%s))) seconds"

    # ── Cleanup ──
    echo
    echo "── Cleanup: uninstall stub add-on ──"
    ssh "${HOST}" sudo ha addons uninstall "${STUB_SLUG}" || true
    ssh "${HOST}" sudo rm -f "/tmp/${STUB_SLUG}.yaml" || true
    echo
    echo "TRANSCRIPT DONE: ${TRANSCRIPT}"
    echo "RESULT: ${RESULT}"
    ```

    Notes on design choices:
    - `exec > >(tee ...)` writes the entire run transcript to the log file while also printing it. The agent can paste this back into 09-SUMMARY.md.
    - `sha256sum` fingerprints protect the transcript from leaking the token values themselves to git (the values NEVER go in the transcript; only the fingerprints). This satisfies PITFALLS S-1 even if the transcript is shared.
    - The 60-second wait after Supervisor restart is empirically enough time on haos-op3050-1; if the token capture step fails with empty values, the agent reruns with a longer sleep.
    - Trap-based cleanup ensures the stub add-on is uninstalled even if the script aborts mid-flow.
    - The `ha addons stdin` path is the most reliable non-interactive way to execute a shell command inside an add-on container on haos-op3050-1's Supervisor install.
    - Service-disruption notice is at the script's header so anyone reading the script (not just the run output) sees the warning.

    #!/usr/bin/env bash
    # spike-pitfalls10-backup-addon-config.sh — Phase 9 PITFALLS §10:
    # does HA's backup integration cover files under /addon_config?
    # Reuses the already-installed `authentik` add-on (per CONTEXT.md D-16).
    #
    # PRE-APPROVED in /gsd-discuss-phase. Read-mostly: the only host action
    # is triggering a backup, which is a brief snapshot.

    set -euo pipefail

    HOST="${1:-haos-op3050-1}"
    AUTHENTIK_SLUG="${2:-authentik}"
    RUN_ID="$(date -u +%Y%m%dT%H%M%SZ)-$$"
    TRANSCRIPT="/tmp/spike-pitfalls10-${RUN_ID}.log"
    SENTINEL_NAME="phase9-sentinel-${RUN_ID}.txt"

    exec > >(tee -a "${TRANSCRIPT}") 2>&1

    echo "PITFALLS §10 SPIKE — Phase 9 backup integration vs /addon_config mount"
    echo "Host: ${HOST}    Authentik slug: ${AUTHENTIK_SLUG}    Transcript: ${TRANSCRIPT}"
    echo

    # ── Step 1: create sentinel inside the authentik add-on container ──
    echo "── Step 1: write sentinel file under /addon_config inside authentik ──"
    ssh "${HOST}" sudo ha addons stdin "${AUTHENTIK_SLUG}" \
        sh -c "echo 'phase-9-sentinel-data' > /addon_config/${SENTINEL_NAME}"
    ssh "${HOST}" sudo ha addons stdin "${AUTHENTIK_SLUG}" \
        sh -c "ls -la /addon_config/${SENTINEL_NAME} && cat /addon_config/${SENTINEL_NAME}"
    echo

    # ── Step 2: trigger HA backup ──
    echo "── Step 2: trigger HA backup integration ──"
    BACKUP_NAME="phase9-backup-${RUN_ID}.tar"
    ssh "${HOST}" sudo ha backups new "${BACKUP_NAME}" || true
    sleep 5
    ssh "${HOST}" sudo ha backups list | grep "${BACKUP_NAME}" || {
        echo "Backup creation failed; check HA logs at 'ha backups list' and 'ha supervisor logs'"
        echo "RESULT: inconclusive"
        exit 1
    }
    echo

    # ── Step 3: pull backup tarball locally and inspect ──
    echo "── Step 3: download backup tarball and inspect contents ──"
    BACKUP_DIR=/tmp/${BACKUP_NAME%.tar}
    mkdir -p "${BACKUP_DIR}"
    rsync -av "${HOST}:/backup/${BACKUP_NAME}" /tmp/
    tar -xf "/tmp/${BACKUP_NAME}" -C "${BACKUP_DIR}"
    echo "   Backup root directory contents (first 30):"
    ls "${BACKUP_DIR}" | head -30
    echo

    # ── Step 4: search for the sentinel under any backup path ──
    echo "── Step 4: search backup for sentinel file ──"
    if grep -r "phase-9-sentinel-data" "${BACKUP_DIR}" 2>/dev/null; then
        RESULT="addon_config_backed_up"
    else
        RESULT="addon_config_not_backed_up"
    fi
    echo "RESULT: ${RESULT}"
    echo

    # ── Step 5: cleanup ──
    echo "── Cleanup: remove sentinel from add-on + remove local backup copy ──"
    ssh "${HOST}" sudo ha addons stdin "${AUTHENTIK_SLUG}" \
        sh -c "rm -f /addon_config/${SENTINEL_NAME}" || true
    rm -rf "${BACKUP_DIR}" /tmp/${BACKUP_NAME}
    echo
    echo "TRANSCRIPT DONE: ${TRANSCRIPT}"
    echo "RESULT: ${RESULT}"
    ```

    Notes on design choices:
    - The script does NOT install or modify the add-on — it reuses the existing authentik install. Per D-16, "No new add-on needed; HA-supervisor-level mount semantics are what we're verifying."
